cmake_minimum_required(VERSION 3.24)
project(xt4_native_detection LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(XT4_ASAN "Address and undefined behavior sanitizers" OFF)
option(XT4_TSAN "Thread sanitizer" OFF)
option(XT4_BUILD_HOST_TOOLS "Build macOS host benchmark tools" ${APPLE})
if(XT4_ASAN AND XT4_TSAN)
  message(FATAL_ERROR "Choose one sanitizer")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
if(XT4_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(XT4_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
# One complete archive; host reference sources are never linked into it.
set(XT4_CORE_SOURCES
  src/arena.cpp src/tensor.cpp src/engine.cpp src/model_data.cpp
  src/prepare.cpp src/geometry.cpp src/context.cpp src/postprocess.cpp
  src/worker.cpp src/result_exchange.cpp src/overlay.cpp src/firmware_adapter.cpp)
add_library(xt4_face STATIC ${XT4_CORE_SOURCES})
target_include_directories(xt4_face PUBLIC include PRIVATE src)
target_compile_options(xt4_face PRIVATE -fno-exceptions -fno-rtti)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(xt4_face PRIVATE -fno-vectorize -fno-slp-vectorize)
else()
  target_compile_options(xt4_face PRIVATE -fno-tree-vectorize -fno-tree-slp-vectorize)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(CMAKE_CROSSCOMPILING)
  target_compile_options(xt4_face PRIVATE -ffreestanding -fno-builtin -fno-stack-protector -ffunction-sections -fdata-sections)
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_fixtures.py
    ${CMAKE_CURRENT_BINARY_DIR}/arm_fixtures.h RESULT_VARIABLE EMBED_RESULT)
  if(NOT EMBED_RESULT EQUAL 0)
    message(FATAL_ERROR "Fixture verification/embedding failed")
  endif()
  execute_process(COMMAND ${CMAKE_CXX_COMPILER} -march=armv7-a -marm -mfloat-abi=soft -print-file-name=libm.a
    OUTPUT_VARIABLE ARM_LIBM OUTPUT_STRIP_TRAILING_WHITESPACE)
  execute_process(COMMAND ${CMAKE_CXX_COMPILER} -march=armv7-a -marm -mfloat-abi=soft -print-libgcc-file-name
    OUTPUT_VARIABLE ARM_LIBGCC OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT EXISTS "${ARM_LIBM}" OR NOT EXISTS "${ARM_LIBGCC}")
    message(FATAL_ERROR "Explicit soft-float math/compiler libraries unavailable")
  endif()
  add_executable(xt4_face_smoke arm/start.S arm/runtime.cpp arm/smoke_main.cpp)
  set_target_properties(xt4_face_smoke PROPERTIES SUFFIX ".elf")
  target_include_directories(xt4_face_smoke PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_compile_options(xt4_face_smoke PRIVATE -ffreestanding -fno-builtin -fno-stack-protector
    $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti;-fno-threadsafe-statics;-fno-tree-vectorize;-fno-tree-slp-vectorize>)
  target_link_options(xt4_face_smoke PRIVATE -nostdlib
    "-Wl,-T,${CMAKE_CURRENT_SOURCE_DIR}/arm/link.ld"
    "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/xt4_face_smoke.map" -Wl,--gc-sections -Wl,-z,noexecstack)
  target_link_libraries(xt4_face_smoke PRIVATE xt4_face -Wl,--start-group "${ARM_LIBM}" "${ARM_LIBGCC}" -Wl,--end-group)
  return()
endif()

enable_testing()
find_package(Threads REQUIRED)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_reference.py
  ${CMAKE_CURRENT_BINARY_DIR}/reference RESULT_VARIABLE REFERENCE_RESULT)
if(NOT REFERENCE_RESULT EQUAL 0)
  message(FATAL_ERROR "Reference source verification failed")
endif()
add_executable(xt4_face_reference tools/reference_main.cpp
  ${CMAKE_CURRENT_BINARY_DIR}/reference/facedetectcnn.cpp
  ${CMAKE_CURRENT_BINARY_DIR}/reference/facedetectcnn-model.cpp
  ${CMAKE_CURRENT_BINARY_DIR}/reference/facedetectcnn-data.cpp)
target_include_directories(xt4_face_reference PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/reference)
target_compile_options(xt4_face_reference PRIVATE -Wno-unused-parameter)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(xt4_face_reference PRIVATE -fno-vectorize -fno-slp-vectorize)
else()
  target_compile_options(xt4_face_reference PRIVATE -fno-tree-vectorize -fno-tree-slp-vectorize)
endif()
foreach(name reference arena prepare geometry engine api postprocess parity worker exchange overlay firmware_adapter)
  add_executable(${name}_test tests/${name}_test.cpp)
  target_include_directories(${name}_test PRIVATE src)
  target_compile_definitions(${name}_test PRIVATE XT4_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" XT4_BINARY_DIR="${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(${name}_test PRIVATE xt4_face Threads::Threads)
  add_test(NAME ${name} COMMAND ${name}_test)
endforeach()
add_executable(c_header_test tests/c_header_test.c)
target_link_libraries(c_header_test PRIVATE xt4_face)
set_target_properties(c_header_test PROPERTIES LINKER_LANGUAGE CXX)
add_test(NAME c_header COMMAND c_header_test)
add_test(NAME arm_smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/arm_smoke_test.py)
add_test(NAME toolchain_lock COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/toolchain_lock_test.py)
add_test(NAME arm_audit COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/arm_audit_test.py)
if(XT4_BUILD_HOST_TOOLS)
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/build_identity.py
    ${CMAKE_CURRENT_BINARY_DIR}/build_identity.h ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION} ${CMAKE_BUILD_TYPE}
    RESULT_VARIABLE IDENTITY_RESULT)
  if(NOT IDENTITY_RESULT EQUAL 0)
    message(FATAL_ERROR "Build identity generation failed")
  endif()
  file(GLOB IDENTITY_INPUTS CONFIGURE_DEPENDS src/*.* include/*.h tools/*.*)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${IDENTITY_INPUTS})
  add_executable(xt4_face_run tools/run_main.cpp tools/report_json.cpp)
  target_include_directories(xt4_face_run PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_compile_definitions(xt4_face_run PRIVATE XT4_BINARY_SIZES_PATH="${CMAKE_CURRENT_BINARY_DIR}/runner-sizes.json")
  target_link_libraries(xt4_face_run PRIVATE xt4_face)
  add_custom_command(TARGET xt4_face_run POST_BUILD COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tools/binary_sizes.py $<TARGET_FILE:xt4_face_run> ${CMAKE_CURRENT_BINARY_DIR}/runner-sizes.json)
  add_executable(runner_test tests/runner_test.cpp)
  target_compile_definitions(runner_test PRIVATE XT4_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" XT4_BINARY_DIR="${CMAKE_CURRENT_BINARY_DIR}")
  add_test(NAME runner COMMAND runner_test)
endif()
